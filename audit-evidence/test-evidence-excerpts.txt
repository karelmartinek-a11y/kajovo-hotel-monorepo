    17	def api_request(
    18	    opener: urllib.request.OpenerDirector,
    19	    base_url: str,
    20	    path: str,
    21	    method: str = "GET",
    22	    payload: dict[str, object] | None = None,
    23	    headers: dict[str, str] | None = None,
    24	) -> tuple[int, dict[str, object] | list[dict[str, object]] | None]:
    25	    url = f"{base_url}{path}"
    26	    data = None
    27	    request_headers = headers.copy() if headers else {}
    28	
    29	    if payload is not None:
    30	        data = json.dumps(payload).encode("utf-8")
    31	        request_headers["Content-Type"] = "application/json"
    32	
    33	    request = urllib.request.Request(url=url, data=data, headers=request_headers, method=method)
    34	    try:
    35	        with opener.open(request, timeout=REQUEST_TIMEOUT_SECONDS) as response:
    36	            raw = response.read().decode("utf-8")
    37	            return response.status, json.loads(raw) if raw else None
    38	    except urllib.error.HTTPError as exc:
    39	        raw = exc.read().decode("utf-8")
    40	        parsed = json.loads(raw) if raw else None
    41	        return exc.code, parsed

    44	def csrf_header(cookie_jar: CookieJar) -> dict[str, str]:
    45	    token = next((cookie.value for cookie in cookie_jar if cookie.name == "kajovo_csrf"), "")
    46	    return {"x-csrf-token": token} if token else {}

   147	async function visibleBrandCount(page: Page): Promise<number> {
   148	  return page.locator('[data-brand-element="true"]').evaluateAll((nodes) =>
   149	    nodes.filter((node) => {
   150	      const element = node as HTMLElement;
   151	      const style = window.getComputedStyle(element);
   152	      const rect = element.getBoundingClientRect();
   153	      return style.visibility !== 'hidden' && style.display !== 'none' && rect.width > 0 && rect.height > 0;
   154	    }).length,
   155	  );
   156	}

   158	async function interactiveCandidates(page: Page): Promise<Locator[]> {
   159	  const selectors = [
   160	    'main button:visible',
   161	    'main a[href]:visible',
   162	    'main select:visible',
   163	    'main input:visible',
   164	    'main textarea:visible',
   165	  ];
   166	  const locators: Locator[] = [];
   167	  for (const selector of selectors) {
   168	    const locator = page.locator(selector);
   169	    const count = await locator.count();
   170	    for (let i = 0; i < Math.min(count, 6); i += 1) {
   171	      locators.push(locator.nth(i));
   172	    }
   173	  }
   174	  return locators;

   177	async function expectElementUnobscured(locator: Locator) {
   178	  await locator.evaluate((element) => {
   179	    element.scrollIntoView({ block: 'center', inline: 'center', behavior: 'instant' });
   180	  });
   181	  const box = await locator.boundingBox();
   182	  if (!box) {
   183	    return;
   184	  }
   185	  const x = box.x + box.width / 2;
   186	  const y = box.y + box.height / 2;
   187	  const unobscured = await locator.evaluate((element, point) => {
   188	    const hit = document.elementFromPoint(point.x, point.y);
   189	    return Boolean(hit && (hit === element || element.contains(hit) || hit.contains(element)));
   190	  }, { x, y });
   191	  expect(unobscured).toBeTruthy();
   192	}

   199	  const overflow = await page.evaluate(() => {
